# The coprocessor firmware: protocol timing, capture, sampling and the power
# path.  Everything with a deadline runs here.
cmake_minimum_required(VERSION 3.13)

# PICO_BOARD defaults to pimoroni_pico_plus2_rp2350 because the SDK (software
# development kit) has no board file for the Waveshare RP2350-CAN module the
# bring-up runs on (RP2350A, 4 MB flash); the default builds and runs on it.
# Override with -DPICO_BOARD=... for another module.  The final board needs an
# RP2350B: servo outputs, servo inputs, the ESC (electronic speed controller)
# channels, the one-wire programming lines, the host link, CAN (Controller
# Area Network), I2C (Inter-Integrated Circuit), the encoder, the load gates
# and the safety line come to 27 to 32 GPIO (general-purpose input/output),
# and a Pico 2 brings out 26.
if(NOT DEFINED PICO_BOARD)
    set(PICO_BOARD pimoroni_pico_plus2_rp2350)
endif()
if(NOT DEFINED PICO_PLATFORM)
    set(PICO_PLATFORM rp2350)
endif()

include(pico_sdk_import.cmake)
project(rcbench-iomcu C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

pico_sdk_init()

# The same directories the panel registers as ESP-IDF (Espressif
# Internet-of-Things Development Framework) components and the host suite
# builds with plain gcc.  One source list, three build systems.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/link"  link)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/bench" bench)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/outputs" outputs)
# The current monitors: drivers, schedule and the service core 1 runs
# (sense_svc.c), on the I2C block through src/sense_i2c.c.  The SENSE page
# validates an INA228 set-up with ina228_calibrate(), so the page and the
# driver hold one rule.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/sense"  sense)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/dshot"  dshot)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/ppm"    ppm)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/safety" safety)
# The move rules (servo_move.c) run in the sensor schedule's capture.  The
# servo procedures beside them are not called here; --gc-sections drops
# them from the image.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/servo"  servo)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/openyge" openyge)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../shared/can"     can)

add_executable(rcbench-iomcu
    src/main.c
    src/xl2515.c
    src/out_pwm.c
    src/out_ppm.c
    src/pd_uart.c
    src/out_dshot.c
    src/outputs_hw.c
    src/out_store.c
    src/sense_i2c.c
    src/sense_core1.c
    src/tone_cap.c
    src/tone_core1.c
    # Generated by tools/gen_board_art.py from artwork/rp2350-can.png, and
    # checked in so this build needs no Python and no image library.
    src/art_rp2350_can.c
)
target_include_directories(rcbench-iomcu PRIVATE include src)

# Core 0's stack: 4096 bytes, all of the SCRATCH_Y region at 0x20081000.  The
# SDK's default is 2048, and the deepest chain from main() with panic() and
# one interrupt on top is 2044 (tools/stack_check.py --iomcu).  The SDK's own
# core 1 stack takes its size from the same definition and fills SCRATCH_X;
# core 1 runs on the array sense_core1.c passes.  The link fails when
# anything else is placed in either region.
target_compile_definitions(rcbench-iomcu PRIVATE PICO_STACK_SIZE=0x1000)

# A bring-up build for the current monitors: with no part enabled in flash,
# both are enabled at the SENSE page's defaults for that boot, not saved.
# Off in every image that is released; docs/FirstRun.md §8 uses it.
option(IOMCU_SENSE_BRINGUP "Enable both current monitors at their defaults" OFF)
if(IOMCU_SENSE_BRINGUP)
    target_compile_definitions(rcbench-iomcu PRIVATE IOMCU_SENSE_BRINGUP=1)
else()
    target_compile_definitions(rcbench-iomcu PRIVATE IOMCU_SENSE_BRINGUP=0)
endif()

# A debug build that prints INA3221 CH1's 1 ms samples on the USB console:
# for 4 s after a changed servo pulse or a capture edge, and for 10 s after
# `t` on the console (shared/sense/sense_trace.h, docs/FirstRun.md 8.9).
# Off in every image that is released: with it off the image holds no trace
# code.
option(SENSE_TRACE "Print INA3221 CH1's 1 ms samples on the console" OFF)
if(SENSE_TRACE)
    target_sources(rcbench-iomcu PRIVATE src/sense_trace_hw.c)
    target_compile_definitions(rcbench-iomcu PRIVATE SENSE_TRACE=1)
else()
    target_compile_definitions(rcbench-iomcu PRIVATE SENSE_TRACE=0)
endif()

# The PIO (programmable input/output) programs are assembled into headers at
# build time, so the .pio file is the only copy of a program and nothing can
# drift from it.  The generated headers land in the build tree.
pico_generate_pio_header(rcbench-iomcu
    "${CMAKE_CURRENT_LIST_DIR}/src/ppm.pio")
pico_generate_pio_header(rcbench-iomcu
    "${CMAKE_CURRENT_LIST_DIR}/src/dshot.pio")
pico_generate_pio_header(rcbench-iomcu
    "${CMAKE_CURRENT_LIST_DIR}/src/pd_uart.pio")
pico_generate_pio_header(rcbench-iomcu
    "${CMAKE_CURRENT_LIST_DIR}/src/tone_cap.pio")

# The capture program as pioasm assembles it must be the program the host
# suite runs (test/host/test_tone_pio.c, fixtures/tone_cap_words.txt).
add_custom_command(TARGET rcbench-iomcu PRE_LINK
    COMMAND ${CMAKE_COMMAND}
            -DHEADER=${CMAKE_CURRENT_BINARY_DIR}/tone_cap.pio.h
            -DWORDS=${CMAKE_CURRENT_LIST_DIR}/../../test/host/fixtures/tone_cap_words.txt
            -DNAME=tone_cap
            -P "${CMAKE_CURRENT_LIST_DIR}/cmake/pio_words.cmake"
    VERBATIM)

# The coprocessor's own sources are held to the same warnings as the host
# build, scoped to this target so the SDK's headers are not judged by them.
# -Wconversion is off here: the SDK's register macros trip it, and the shared
# arithmetic that needs it is held to it by the host suite.
target_compile_options(rcbench-iomcu PRIVATE
    -Wall -Wextra -Werror -Wshadow)
target_link_libraries(rcbench-iomcu
    pico_stdlib
    hardware_pio
    hardware_pwm
    hardware_dma
    hardware_spi
    hardware_gpio
    hardware_flash
    hardware_sync
    hardware_i2c
    # Core 1 runs the sensor bus; flash_safe_execute() parks it for the
    # store's windows through the multicore lock-out.
    pico_multicore
    pico_flash
    rcbench_link
    rcbench_sense
    rcbench_outputs
    rcbench_dshot
    rcbench_ppm
    rcbench_bench
    rcbench_safety
    rcbench_servo
    rcbench_openyge
    rcbench_can
)
#
# Diagnostics go out over USB (Universal Serial Bus) CDC (communications
# device class), not over a UART (universal asynchronous
# receiver-transmitter).  The SDK's default is stdio on uart0; both lines are
# stated explicitly so no diagnostic output lands on a UART pin.
#
pico_enable_stdio_usb(rcbench-iomcu 1)
pico_enable_stdio_uart(rcbench-iomcu 0)

pico_add_extra_outputs(rcbench-iomcu)

# The image has to fit the module fitted, not the one the board file names.
# out_store.c pins its two sectors to the top of 4 MB for the same reason;
# this is the other half of that, and it is checked after the link because a
# binary's size is not a compile-time constant. 4 MB - 8 kB = 4186112.
add_custom_command(TARGET rcbench-iomcu POST_BUILD
    COMMAND ${CMAKE_COMMAND}
            -DIMAGE=$<TARGET_FILE_DIR:rcbench-iomcu>/rcbench-iomcu.bin
            -DLIMIT=4186112
            -P "${CMAKE_CURRENT_LIST_DIR}/cmake/image_fits.cmake"
    VERBATIM)
